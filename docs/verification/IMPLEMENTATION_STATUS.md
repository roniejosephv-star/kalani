  ```sh

    "$SIMULATOR_UDID" "$DISPOSABLE_TOKEN_FILE" "$TAMIL_M4A"

    "$SIMULATOR_UDID" "$DISPOSABLE_TOKEN_FILE" "$ENGLISH_M4A"
  ```
